// 命令 server 启动课下作业平台 HTTP 服务。
package main

import (
        "log"
        "net/http"
        "time"

        "onlineclass/internal/clock"
        "onlineclass/internal/config"
        "onlineclass/internal/httpapi"
        "onlineclass/internal/store"
)

func main() {
        cfg, err := config.Load()
        if err != nil {
                log.Fatalf("配置错误: %v", err)
        }

        st, err := store.Open(cfg.DBPath)
        if err != nil {
                log.Fatalf("打开数据库失败: %v", err)
        }
        defer func() { _ = st.Close() }()

        if err := st.Migrate(); err != nil {
                log.Fatalf("数据库迁移失败: %v", err)
        }
        if cfg.Seed {
                if err := st.Seed(); err != nil {
                        log.Fatalf("写入演示数据失败: %v", err)
                }
        }

        clk := clock.New(cfg.ClockOffset)
        app := httpapi.New(st, clk, 8*time.Hour)

        srv := &http.Server{
                Addr:              ":" + cfg.Port,
                Handler:           app.Handler(),
                ReadHeaderTimeout: 5 * time.Second,
        }
        log.Printf("课下作业平台启动：http://localhost:%s 数据库=%s 教学时钟偏移=%s",
                cfg.Port, cfg.DBPath, clk.Offset())
        if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
                log.Fatalf("服务退出: %v", err)
        }
}

// Package apperr 定义结构化业务错误。
//
// 约定：业务拒绝一律用 4xx + 稳定错误码，handler 只做映射，不自行编造状态码。
package apperr

import "fmt"

// Error 是可映射为 HTTP 响应的业务错误。
type Error struct {
        Status  int
        Code    string
        Message string
        Details map[string]any
}

func (e *Error) Error() string {
        if len(e.Details) == 0 {
                return fmt.Sprintf("%s: %s", e.Code, e.Message)
        }
        return fmt.Sprintf("%s: %s %v", e.Code, e.Message, e.Details)
}

// New 构造业务错误。
func New(status int, code, message string) *Error {
        return &Error{Status: status, Code: code, Message: message}
}

// WithDetails 附加结构化上下文（如当前版本号），返回自身便于链式调用。
func (e *Error) WithDetails(kv map[string]any) *Error {
        if e.Details == nil {
                e.Details = make(map[string]any, len(kv))
        }
        for k, v := range kv {
                e.Details[k] = v
        }
        return e
}

// 常用错误构造器。
func Forbidden(message string) *Error {
        return New(403, "forbidden_role", message)
}

func NotFound(message string) *Error {
        return New(404, "not_found", message)
}

func Invalid(message string) *Error {
        return New(400, "invalid_input", message)
}

func Conflict(code, message string) *Error {
        return New(409, code, message)
}

// Package auth 提供口令哈希与校验。
//
// 采用 PBKDF2-HMAC-SHA256（标准库实现，无第三方依赖）。教学演示账号口令统一，
// 但存储形式与生产一致：只存盐与派生值，不存明文。
package auth

import (
        "crypto/hmac"
        "crypto/rand"
        "crypto/sha256"
        "crypto/subtle"
        "encoding/base64"
        "errors"
        "fmt"
        "strconv"
        "strings"
)

const (
        iterations = 120000
        saltLen    = 16
        keyLen     = 32
        scheme     = "pbkdf2-sha256"
)

// ErrInvalidHash 表示存储的哈希格式非法。
var ErrInvalidHash = errors.New("口令哈希格式非法")

// HashPassword 生成随机盐并返回可存库的编码串：pbkdf2-sha256$iter$salt$key。
func HashPassword(password string) (string, error) {
        if password == "" {
                return "", errors.New("口令不能为空")
        }
        salt := make([]byte, saltLen)
        if _, err := rand.Read(salt); err != nil {
                return "", fmt.Errorf("生成盐失败: %w", err)
        }
        key := pbkdf2SHA256([]byte(password), salt, iterations, keyLen)
        return fmt.Sprintf("%s$%d$%s$%s", scheme, iterations,
                base64.RawStdEncoding.EncodeToString(salt),
                base64.RawStdEncoding.EncodeToString(key)), nil
}

// VerifyPassword 校验明文口令与编码串是否匹配；使用常量时间比较。
func VerifyPassword(encoded, password string) bool {
        parts := strings.Split(encoded, "$")
        if len(parts) != 4 || parts[0] != scheme {
                return false
        }
        iter, err := strconv.Atoi(parts[1])
        if err != nil || iter <= 0 {
                return false
        }
        salt, err := base64.RawStdEncoding.DecodeString(parts[2])
        if err != nil {
                return false
        }
        want, err := base64.RawStdEncoding.DecodeString(parts[3])
        if err != nil || len(want) == 0 {
                return false
        }
        got := pbkdf2SHA256([]byte(password), salt, iter, len(want))
        return subtle.ConstantTimeCompare(got, want) == 1
}

// pbkdf2SHA256 是 PBKDF2（RFC 8018）在 HMAC-SHA256 上的最小实现。
func pbkdf2SHA256(password, salt []byte, iter, keyLen int) []byte {
        prf := hmac.New(sha256.New, password)
        hashLen := prf.Size()
        blocks := (keyLen + hashLen - 1) / hashLen
        out := make([]byte, 0, blocks*hashLen)

        buf := make([]byte, 4)
        for block := 1; block <= blocks; block++ {
                prf.Reset()
                prf.Write(salt)
                buf[0] = byte(block >> 24)
                buf[1] = byte(block >> 16)
                buf[2] = byte(block >> 8)
                buf[3] = byte(block)
                prf.Write(buf)

                u := prf.Sum(nil)
                t := make([]byte, len(u))
                copy(t, u)

                for i := 1; i < iter; i++ {
                        prf.Reset()
                        prf.Write(u)
                        u = prf.Sum(u[:0])
                        for j := range t {
                                t[j] ^= u[j]
                        }
                }
                out = append(out, t...)
        }
        return out[:keyLen]
}

// Package clock 提供可注入的业务时钟。
//
// 业务代码一律通过 Clock.Now() 取当前时间，禁止直接调用 time.Now()：
// 截止判定、发奖时间与事件时间必须同源，测试与教学演示才能复现。
package clock

import "time"

// Clock 是带固定偏移的业务时钟，并发安全（只读字段）。
type Clock struct {
        offset time.Duration
}

// New 创建时钟。offset 为教学演示偏移，0 表示与系统时间一致。
func New(offset time.Duration) *Clock { return &Clock{offset: offset} }

// Now 返回 UTC 时间。存储与判定统一使用 UTC，展示层再转换时区。
func (c *Clock) Now() time.Time { return time.Now().UTC().Add(c.offset) }

// Offset 返回当前偏移量，用于页面与日志标注"使用了教学时钟"。
func (c *Clock) Offset() time.Duration { return c.offset }

// At 返回相对现在偏移 d 的时间，便于计算"截止时间"等派生值。
func (c *Clock) At(d time.Duration) time.Time { return c.Now().Add(d) }

package config

// Config 是服务运行所需的全部配置。
type Config struct {
        // Port 监听端口，默认 8080。
        Port string
        // DBPath SQLite 数据库文件路径，默认 data/app.db。
        DBPath string
        // ClockOffset 教学时钟偏移，用于演示逾期与补交；默认 0。
        ClockOffset time.Duration
        // Seed 是否在启动时写入预置演示数据，默认 true。
        Seed bool
}

// Load 从环境变量加载配置并校验。
//
// 环境变量：
//
//        PORT                   监听端口
//        DB_PATH                数据库文件
//        TEACHING_CLOCK_OFFSET  教学时钟偏移（如 2h、-30m、90s）
//        SEED                   1/0，是否写入演示数据
func Load() (Config, error) {
        cfg := Config{
                Port:   env("PORT", "8080"),
                DBPath: env("DB_PATH", "data/app.db"),
                Seed:   env("SEED", "1") == "1",
        }
        if cfg.Port == "" {
                return cfg, fmt.Errorf("PORT 不能为空")
        }
        if cfg.DBPath == "" {
                return cfg, fmt.Errorf("DB_PATH 不能为空")
        }
        raw := env("TEACHING_CLOCK_OFFSET", "")
        if raw != "" {
                d, err := time.ParseDuration(raw)
                if err != nil {
                        return cfg, fmt.Errorf("TEACHING_CLOCK_OFFSET 格式错误（示例 2h、-30m）: %w", err)
                }
                cfg.ClockOffset = d
        }
        return cfg, nil
}

// env 读取环境变量；未设置或设为空串时回落默认值（空串等同未设置，便于
// 容器与 CI 中以空值占位而不破坏默认配置）。
func env(key, def string) string {
        if v, ok := os.LookupEnv(key); ok && v != "" {
                return v
        }
        return def
}

// handleMe 返回当前身份与可执行动作（用于演示"身份只来自会话"）。
func (s *Server) handleMe(w http.ResponseWriter, r *http.Request) {
        u, _ := userFrom(r)
        actions := []Action{ActionViewAssignment, ActionSubmit, ActionReview, ActionGrantExtension, ActionRecordReminder, ActionViewStudent}
        allowed := make([]string, 0, len(actions))
        for _, a := range actions {
                if Allow(u.Role, a, true, true) {

                }
        }
        writeJSON(w, http.StatusOK, map[string]any{
                "id":          u.ID,
                "name":        u.Name,
                "role":        u.Role,
                "class_id":    u.ClassID,
                "permissions": allowed,
        })
}

// assignmentFor 载入路径中的作业并做班级隔离。
//
// 不存在与跨班都返回 404：避免通过状态码差异判断作业是否存在。
func (s *Server) assignmentFor(w http.ResponseWriter, r *http.Request) (store.Assignment, store.User, bool) {
        u, ok := userFrom(r)
        if !ok {
                writeAPIError(w, http.StatusUnauthorized, "unauthenticated", "请先登录")
                return store.Assignment{}, store.User{}, false
        }
        a, err := s.store.GetAssignment(r.Context(), r.PathValue("id"))
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        writeAPIError(w, http.StatusNotFound, "not_found", "作业不存在")
                        return store.Assignment{}, store.User{}, false
                }
                logError("查询作业", err)
                writeAPIError(w, http.StatusInternalServerError, "internal", "服务内部错误")
                return store.Assignment{}, store.User{}, false
        }
        if a.ClassID != u.ClassID {
                writeAPIError(w, http.StatusNotFound, "not_found", "作业不存在")
                return store.Assignment{}, store.User{}, false
        }
        return a, u, true
}

// handleAssignmentGet 返回作业的对外视图（已发布以快照为准）。
func (s *Server) handleAssignmentGet(w http.ResponseWriter, r *http.Request) {
        a, u, ok := s.assignmentFor(w, r)
        if !ok {
                return
        }
        if !Allow(u.Role, ActionViewAssignment, true, false) {
                writeAPIError(w, http.StatusForbidden, "forbidden_role", "当前角色无权查看作业")
                return
        }
        // 草稿仅本班教师可见：对学生与助教按"不存在"处理，避免提前泄露未发布内容。
        if !a.Published() && u.Role != store.RoleTeacher {
                writeAPIError(w, http.StatusNotFound, "not_found", "作业不存在")
                return
        }
        writeJSON(w, http.StatusOK, map[string]any{
                "id":           a.ID,
                "class_id":     a.ClassID,
                "state":        a.State,
                "title":        a.EffectiveTitle(),
                "requirement":  a.EffectiveRequirement(),
                "deadline_at":  a.EffectiveDeadline().Format("2006-01-02T15:04:05Z07:00"),
                "published_at": a.PublishedAt.Format("2006-01-02T15:04:05Z07:00"),
                "can_submit":   Allow(u.Role, ActionSubmit, true, true),
                "can_review":   Allow(u.Role, ActionReview, true, false),
        })
}

// assignmentPayload 是创建/修改草稿的请求体。
type assignmentPayload struct {
        Title       string `json:"title"`
        Requirement string `json:"requirement"`
        DeadlineAt  string `json:"deadline_at"`
}

func (p assignmentPayload) toInput() (service.AssignmentInput, error) {
        in := service.AssignmentInput{Title: p.Title, Requirement: p.Requirement}
        if p.DeadlineAt == "" {
                return in, nil
        }
        t, err := time.Parse(time.RFC3339, p.DeadlineAt)
        if err != nil {
                return in, apperr.Invalid("deadline_at 必须是 RFC3339 时间，例如 2026-12-01T23:59:00+08:00")
        }
        in.DeadlineAt = t
        return in, nil
}

// decodeAssignment 解析请求体；未知字段视为参数错误，避免前端拼错字段被静默忽略。
func decodeAssignment(r *http.Request) (service.AssignmentInput, error) {
        var p assignmentPayload
        dec := json.NewDecoder(io.LimitReader(r.Body, 64<<10))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&p); err != nil {
                return service.AssignmentInput{}, apperr.Invalid("请求体格式错误，需要 JSON：title、requirement、deadline_at")
        }
        return p.toInput()
}

func assignmentView(a store.Assignment) map[string]any {
        v := map[string]any{
                "id":          a.ID,
                "class_id":    a.ClassID,
                "state":       a.State,
                "title":       a.EffectiveTitle(),
                "requirement": a.EffectiveRequirement(),
                "deadline_at": a.EffectiveDeadline().Format(time.RFC3339),
                "snapshot_by": a.SnapshotBy,
        }
        if !a.PublishedAt.IsZero() {
                v["published_at"] = a.PublishedAt.Format(time.RFC3339)
        }
        if !a.SnapshotAt.IsZero() {
                v["snapshot_at"] = a.SnapshotAt.Format(time.RFC3339)
        }
        return v
}

// writeServiceError 把业务错误映射为响应；未识别的错误按 500 处理并记录日志。
func writeServiceError(w http.ResponseWriter, err error) {
        var ae *apperr.Error
        if errors.As(err, &ae) {
                body := map[string]any{"code": ae.Code, "message": ae.Message}
                if len(ae.Details) > 0 {
                        body["details"] = ae.Details
                }
                writeJSON(w, ae.Status, body)
                return
        }
        logError("业务处理", err)
        writeAPIError(w, http.StatusInternalServerError, "internal", "服务内部错误")
}

// handleLogin 校验账号口令并建立会话。
//
// 失败一律返回同一条提示，避免通过错误信息探测账号是否存在。
func (s *Server) handleLogin(w http.ResponseWriter, r *http.Request) {
        if err := r.ParseForm(); err != nil {
                s.renderLogin(w, http.StatusBadRequest, "表单格式错误", "")
                return
        }
        userID := strings.TrimSpace(r.PostFormValue("user_id"))
        password := r.PostFormValue("password")
        if userID == "" || password == "" {
                s.renderLogin(w, http.StatusBadRequest, "请输入账号与口令", userID)
                return
        }

        u, err := s.store.GetUser(r.Context(), userID)
        if err != nil {
                if !errors.Is(err, store.ErrNotFound) {
                        logError("查询账号", err)
                }
                s.renderLogin(w, http.StatusUnauthorized, "账号或口令不正确", userID)
                return
        }
        if !auth.VerifyPassword(u.PasswordHash, password) {
                s.renderLogin(w, http.StatusUnauthorized, "账号或口令不正确", userID)
                return
        }

        sid, err := s.sessions.Create(u.ID)
        if err != nil {
                logError("创建会话", err)
                http.Error(w, "服务内部错误", http.StatusInternalServerError)
                return
        }
        http.SetCookie(w, &http.Cookie{
                Name:     sessionCookieName,
                Value:    sid,
                Path:     "/",
                HttpOnly: true,
                SameSite: http.SameSiteLaxMode,
                MaxAge:   int((8 * time.Hour).Seconds()),
                // 教学演示走 http，故不设 Secure；部署到 https 时应开启。
        })
        http.Redirect(w, r, "/", http.StatusSeeOther)
}

// handleLogout 销毁会话并清除 Cookie。
func (s *Server) handleLogout(w http.ResponseWriter, r *http.Request) {
        if c, err := r.Cookie(sessionCookieName); err == nil {
                s.sessions.Destroy(c.Value)
        }
        http.SetCookie(w, &http.Cookie{
                Name:     sessionCookieName,
                Value:    "",
                Path:     "/",
                HttpOnly: true,
                SameSite: http.SameSiteLaxMode,
                MaxAge:   -1,
        })
        http.Redirect(w, r, "/login", http.StatusSeeOther)
}

func (s *Server) renderLogin(w http.ResponseWriter, status int, errMsg, userID string) {
        data := map[string]any{"Error": errMsg, "UserID": userID}
        w.Header().Set("Content-Type", "text/html; charset=utf-8")
        w.WriteHeader(status)
        if err := s.tpl.ExecuteTemplate(w, "login.html", data); err != nil {
                logError("渲染登录页", err)
        }
}

// Action 是可授权的操作类型。
type Action string

// 授权动作常量。
const (
        ActionViewAssignment Action = "view_assignment"
        ActionSubmit         Action = "submit"
        ActionReview         Action = "review"
        ActionGrantExtension Action = "grant_extension"
        ActionRecordReminder Action = "record_reminder"
        ActionViewStudent    Action = "view_student"
)

// Allow 判断某角色能否对目标资源执行动作。
//
// 前置：sameClass 表示目标资源与请求者同班；isSelf 表示目标学生就是请求者本人。
// 跨班一律拒绝（由调用方转成 404，避免泄露资源是否存在）。
func Allow(role string, action Action, sameClass, isSelf bool) bool {
        if !sameClass {
                return false
        }
        switch action {
        case ActionViewAssignment:
                // 同班的教师、助教、学生均可查看作业。
                return role == store.RoleTeacher || role == store.RoleAssistant || role == store.RoleStudent
        case ActionSubmit:
                return role == store.RoleStudent && isSelf
        case ActionReview, ActionGrantExtension:
                return role == store.RoleTeacher
        case ActionRecordReminder:
                return role == store.RoleAssistant
        case ActionViewStudent:
                if role == store.RoleTeacher || role == store.RoleAssistant {
                        return true
                }
                return role == store.RoleStudent && isSelf
        }
        return false
}

type ctxKey int

const userCtxKey ctxKey = iota

// withUser 从会话解析当前用户并放入请求上下文。
//
// 身份只来自服务端会话：请求体、查询参数或请求头中的用户编号一律不采信，
// 因此"改参数冒充他人"无法越权。
func (s *Server) withUser(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                cookie, err := r.Cookie(sessionCookieName)
                if err != nil {
                        next.ServeHTTP(w, r)
                        return
                }
                sess, ok := s.sessions.Get(cookie.Value)
                if !ok {
                        next.ServeHTTP(w, r)
                        return
                }
                u, err := s.store.GetUser(r.Context(), sess.UserID)
                if err != nil {
                        // 账号被删除或会话指向无效用户：按未登录处理。
                        next.ServeHTTP(w, r)
                        return
                }
                next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), userCtxKey, u)))
        })
}

// userFrom 返回上下文中的当前用户。
func userFrom(r *http.Request) (store.User, bool) {
        u, ok := r.Context().Value(userCtxKey).(store.User)
        return u, ok
}

// requireAuth 拦截未登录请求，统一返回 401 JSON（页面路由不走它）。
func (s *Server) requireAuth(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if _, ok := userFrom(r); !ok {
                        writeAPIError(w, http.StatusUnauthorized, "unauthenticated", "请先登录")
                        return
                }
                next.ServeHTTP(w, r)
        })
}

                view := studentStatusView(st.StudentStatus)

                if !st.FirstSubmittedAt.IsZero() {
                        view["first_submitted_at"] = st.FirstSubmittedAt.Format(time.RFC3339)
                }
                if st.Extension != nil {
                        view["extension"] = map[string]any{
                                "deadline_at": st.Extension.DeadlineAt.Format(time.RFC3339),

                        msg += "，本次发放 " + strconv.Itoa(n) + " 分"
                }
        }
        return msg
}

        }
        list, err := s.store.ListAssignmentsByClass(r.Context(), u.ClassID)
        if err != nil {

                return
        }

        cards := make([]assignmentCard, 0, len(list))
        for _, a := range list {
                if u.Role != store.RoleTeacher && !a.Published() {
                        continue // 草稿对学生/助教不可见
                }
                cards = append(cards, assignmentCard{
                        ID:          a.ID,
                        Title:       a.EffectiveTitle(),
                        State:       a.State,
                        StateLabel:  labelOf(assignmentStateLabels, a.State),
                        Deadline:    a.EffectiveDeadline().Format("2006-01-02 15:04"),
                        PublishedAt: formatPageTime(a.PublishedAt),
                        IsDraft:     a.State == store.AssignmentDraft,
                        CanReview:   a.Published(),
                })
        }

        data := assignmentsPageData{
                Base:         s.baseData(u),
                Assignments:  cards,
                IsTeacher:    u.Role == store.RoleTeacher,
                DefaultTime:  s.clk.Now().Add(24 * time.Hour).Format("2006-01-02T15:04"),
                FormAction:   "/assignments",
                EditDeadline: s.clk.Now().Add(24 * time.Hour).Format("2006-01-02T15:04"),
        }
        if edit := r.URL.Query().Get("edit"); edit != "" {
                if u.Role != store.RoleTeacher {
                        s.renderPageError(w, r, apperr.Forbidden("只有教师可以编辑作业"), "/assignments")
                        return
                }
                a, err := s.assignmentForPage(r, u, edit)
                if err != nil {
                        s.renderPageError(w, r, err, "/assignments")
                        return
                }
                if a.State != store.AssignmentDraft {
                        s.renderPageError(w, r, apperr.Conflict("assignment_locked", "作业已发布，内容已锁定，不能修改"), "/assignments")
                        return
                }
                data.EditID = a.ID
                data.EditTitle = a.Title
                data.EditRequirement = a.Requirement
                data.EditDeadline = a.DeadlineAt.In(time.Local).Format("2006-01-02T15:04")
                data.FormAction = "/assignments/" + a.ID
                data.Editing = true
        }
        data.Base.OK = okMessage(r)
        s.renderPage(w, http.StatusOK, "assignments.html", data)
}

                row := versionRow{No: v.VersionNo, Content: v.Content, CreatedAt: formatPageTime(v.CreatedAt)}
                if rev, ok := h.Reviews[v.VersionNo]; ok {
                        row.Decision = labelOf(decisionLabels, rev.Decision)
                        row.Feedback = rev.Feedback
                        row.ReviewedAt = formatPageTime(rev.CreatedAt)
                }
                versions = append(versions, row)
        }

        points, err := s.store.TotalPoints(r.Context(), u.ID)
        if err != nil {
                s.renderPageError(w, r, err, "/assignments")
                return
        }
        entries, err := s.store.ListPointsByAssignment(r.Context(), id)
        if err != nil {
                s.renderPageError(w, r, err, "/assignments")
                return
        }
        own := make([]store.PointsEntry, 0, 1)
        for _, e := range entries {
                if e.StudentID == u.ID {
                        own = append(own, e)
                }
        }

        now := s.clk.Now()
        blocked := ""
        switch {
        case h.State == store.StatePassed:
                blocked = "已通过，无需再提交"
        case now.After(h.EffectiveDeadline):
                blocked = "已超过有效截止时间（" + h.EffectiveDeadline.Format("2006-01-02 15:04") + "）"
        }

        data := studentPageData{
                Base:             s.baseData(u),
                Assignment:       a,
                Title:            a.EffectiveTitle(),
                Requirement:      a.EffectiveRequirement(),
                Deadline:         h.EffectiveDeadline.Format("2006-01-02 15:04"),
                State:            h.State,
                StateLabel:       labelOf(stateLabels, h.State),
                CurrentVersionNo: h.CurrentVersionNo,
                Versions:         versions,
                Points:           points,
                Entries:          own,
                RequestID:        newPageRequestID(),
                CanSubmit:        blocked == "",
                BlockedReason:    blocked,
        }
        if n := len(h.Versions); n > 0 {
                data.DraftContent = h.Versions[n-1].Content
        }
        if h.ExtensionApplied {
                if ext, err := s.store.GetExtension(r.Context(), id, u.ID); err == nil {
                        extCopy := ext
                        data.Extension = &extCopy
                }
        }
        data.Base.OK = okMessage(r)
        s.renderPage(w, http.StatusOK, "assignment.html", data)
}

        id := r.PathValue("id")

        for _, st := range board.Students {
                row := reviewRow{
                        StudentID:  st.StudentID,
                        Name:       st.Name,

                        s.renderPageError(w, r, err, "/assignments/"+id+"/review")
                        return

                        latest := h.Versions[n-1]
                        row.VersionID = latest.ID
                        row.VersionNo = latest.VersionNo
                        row.Content = latest.Content
                        row.SubmittedAt = formatPageTime(latest.CreatedAt)

                                row.Decision = labelOf(decisionLabels, rev.Decision)

                                extCopy := ext
                                row.Extension = &extCopy
                        }
                }
                row.CanReview = st.State == store.StateSubmitted && row.VersionID != ""
                if row.CanReview {
                        pending++
                }
                if st.State == store.StatePassed {
                        done++
                }
                rows = append(rows, row)
        }

        data := reviewPageData{
                Base:        s.baseData(u),
                Assignment:  board.Assignment,
                Title:       board.Assignment.EffectiveTitle(),
                Deadline:    board.Assignment.EffectiveDeadline().Format("2006-01-02 15:04"),
                Rows:        rows,
                Pending:     pending,
                Done:        done,
                DefaultTime: s.clk.Now().Add(24 * time.Hour).Format("2006-01-02T15:04"),
        }
        for _, row := range rows {
                if row.State != store.StatePassed && row.Extension == nil {
                        data.Extendable = append(data.Extendable, row)
                }
        }
        data.Base.OK = okMessage(r)
        s.renderPage(w, http.StatusOK, "review.html", data)

func formatPageTime(t time.Time) string {
        if t.IsZero() {
                return ""
        }
        return t.Format("2006-01-02 15:04")
}

// parseFormDeadline 接受浏览器 datetime-local 与本机时区；带时区的 RFC3339 也接受。
func parseFormDeadline(v string) (time.Time, error) {
        v = strings.TrimSpace(v)
        if v == "" {
                return time.Time{}, apperr.Invalid("截止时间不能为空")
        }
        if t, err := time.ParseInLocation("2006-01-02T15:04", v, time.Local); err == nil {
                return t, nil
        }
        if t, err := time.Parse(time.RFC3339, v); err == nil {
                return t, nil
        }
        return time.Time{}, apperr.Invalid("截止时间格式不正确，应为 YYYY-MM-DDTHH:MM")
}

func newPageRequestID() string {
        buf := make([]byte, 8)
        if _, err := rand.Read(buf); err != nil {
                return strconv.FormatInt(time.Now().UnixNano(), 36)
        }
        return hex.EncodeToString(buf)
}

func reviewView(rev store.Review) map[string]any {
        view := map[string]any{
                "id":         rev.ID,
                "version_id": rev.VersionID,
                "decision":   rev.Decision,
                "feedback":   rev.Feedback,
                "teacher_id": rev.TeacherID,
                "created_at": rev.CreatedAt.Format(time.RFC3339),
        }
        if rev.RequestID != "" {
                view["request_id"] = rev.RequestID
        }
        return view
}

// handleReviewPost 教师批阅指定版本：通过 → PASSED，退回 → RETURNED。
func (s *Server) handleReviewPost(w http.ResponseWriter, r *http.Request) {
        u, _ := userFrom(r)
        if u.Role != store.RoleTeacher {
                writeAPIError(w, http.StatusForbidden, "forbidden_role", "只有教师可以批阅")
                return
        }
        if _, _, ok := s.assignmentFor(w, r); !ok {
                return
        }

        var p reviewPayload
        dec := json.NewDecoder(io.LimitReader(r.Body, 64<<10))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&p); err != nil {
                writeServiceError(w, apperr.Invalid("请求体格式错误，需要 JSON：version_id、decision、feedback"))
                return
        }

        res, err := s.reviews.Review(r.Context(), u, r.PathValue("id"), service.ReviewInput{
                VersionID: p.VersionID,
                Decision:  p.Decision,
                Feedback:  p.Feedback,
                RequestID: p.RequestID,
        })
        if err != nil {
                writeServiceError(w, err)
                return
        }

        body := reviewView(res.Review)
        body["state"] = res.Enrollment.State
        body["current_version_no"] = res.Enrollment.CurrentVersionNo
        body["replayed"] = res.Replayed
        body["points_awarded"] = res.PointsAwarded
        body["points"] = res.Points
        status := http.StatusCreated
        if res.Replayed {
                status = http.StatusOK
        }
        writeJSON(w, status, body)
}

// submissionPayload 是学生提交的请求体。
type submissionPayload struct {
        Content     string `json:"content"`
        RequestID   string `json:"request_id"`

func submissionView(sub store.Submission) map[string]any {
        return map[string]any{
                "id":           sub.ID,
                "version_no":   sub.VersionNo,
                "content":      sub.Content,
                "content_hash": sub.ContentHash,
                "request_id":   sub.RequestID,
                "created_at":   sub.CreatedAt.Format(time.RFC3339),
        }

// handleSubmissionPost 学生提交或更新提交（追加新版本）。
func (s *Server) handleSubmissionPost(w http.ResponseWriter, r *http.Request) {
        u, _ := userFrom(r)
        if u.Role != store.RoleStudent {
                writeAPIError(w, http.StatusForbidden, "forbidden_role", "只有学生可以提交作业")
                return
        }
        // 先做资源解析与班级隔离（跨班 404），再校验请求体：
        // 否则"外班 + 空请求体"会先暴露 400，等同于泄露作业存在性。
        if _, _, ok := s.assignmentFor(w, r); !ok {
                return
        }

        var p submissionPayload
        dec := json.NewDecoder(io.LimitReader(r.Body, 64<<10))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&p); err != nil {
                writeServiceError(w, apperr.Invalid("请求体格式错误，需要 JSON：content、request_id"))
                return
        }

        res, err := s.submissions.Submit(r.Context(), u, r.PathValue("id"), service.SubmitInput{
                Content:     p.Content,
                RequestID:   p.RequestID,
                BaseVersion: p.BaseVersion,
        })
        if err != nil {
                writeServiceError(w, err)
                return
        }

        body := submissionView(res.Version)
        body["state"] = res.Enrollment.State
        body["current_version_no"] = res.Enrollment.CurrentVersionNo
        body["effective_deadline"] = res.EffectiveDeadline.Format(time.RFC3339)
        body["extension_applied"] = res.ExtensionApplied
        body["replayed"] = res.Replayed
        // 重放不产生新资源，按 200 返回；新建版本仍是 201。
        status := http.StatusCreated
        if res.Replayed {
                status = http.StatusOK
        }
        writeJSON(w, status, body)
}

// handleSubmissionList 返回某学生在该作业上的全部版本（含历史版本）。
//
// 学生默认查自己且只能查自己；教师/助教需指定 ?student_id= 且限本班。
func (s *Server) handleSubmissionList(w http.ResponseWriter, r *http.Request) {
        u, _ := userFrom(r)

        studentID := r.URL.Query().Get("student_id")
        if u.Role == store.RoleStudent && studentID != "" && studentID != u.ID {
                writeAPIError(w, http.StatusForbidden, "forbidden_role", "学生只能查看自己的提交")
                return
        }

        h, err := s.submissions.History(r.Context(), u, r.PathValue("id"), studentID)
        if err != nil {
                writeServiceError(w, err)
                return
        }

        versions := make([]map[string]any, 0, len(h.Versions))
        for _, sub := range h.Versions {
                view := submissionView(sub)
                if rev, ok := h.Reviews[sub.VersionNo]; ok {
                        view["review"] = reviewView(rev)
                }
                versions = append(versions, view)
        }
        writeJSON(w, http.StatusOK, map[string]any{
                "assignment_id":      h.AssignmentID,
                "student_id":         h.StudentID,
                "state":              h.State,
                "current_version_no": h.CurrentVersionNo,
                "effective_deadline": h.EffectiveDeadline.Format(time.RFC3339),
                "extension_applied":  h.ExtensionApplied,
                "versions":           versions,
        })
}

// AssignmentInput 是创建/修改草稿的入参。
type AssignmentInput struct {
        Title       string
        Requirement string
        DeadlineAt  time.Time
}

// AssignmentService 负责作业的创建、草稿修改与发布。
type AssignmentService struct {
        store *store.Store
        clk   *clock.Clock
}

// NewAssignmentService 构造作业服务。
func NewAssignmentService(st *store.Store, clk *clock.Clock) *AssignmentService {
        return &AssignmentService{store: st, clk: clk}
}

// 校验长度按"字符数"而非字节数，避免中文被误判超长。
const (
        maxTitleRunes       = 100
        maxRequirementRunes = 2000
)

// Create 创建草稿。仅本班教师可操作，班级与作者取自会话身份，不信任请求体。
func (s *AssignmentService) Create(ctx context.Context, actor store.User, in AssignmentInput) (store.Assignment, error) {
        if actor.Role != store.RoleTeacher {
                return store.Assignment{}, apperr.Forbidden("只有教师可以创建作业")
        }
        title, requirement, deadline, err := s.validate(in)
        if err != nil {
                return store.Assignment{}, err
        }

        id, err := newAssignmentID()
        if err != nil {
                return store.Assignment{}, err
        }
        a := store.Assignment{
                ID:          id,
                ClassID:     actor.ClassID,
                TeacherID:   actor.ID,
                Title:       title,
                Requirement: requirement,
                DeadlineAt:  deadline,
                State:       store.AssignmentDraft,
                CreatedAt:   s.clk.Now(),
        }
        if err := s.store.CreateAssignment(ctx, a); err != nil {
                return store.Assignment{}, err
        }
        return a, nil
}

// UpdateDraft 修改草稿。已发布作业拒绝修改（409 assignment_locked），不做静默变更。
func (s *AssignmentService) UpdateDraft(ctx context.Context, actor store.User, id string, in AssignmentInput) (store.Assignment, error) {
        if actor.Role != store.RoleTeacher {
                return store.Assignment{}, apperr.Forbidden("只有教师可以修改作业")
        }
        current, err := s.loadForActor(ctx, actor, id)
        if err != nil {
                return store.Assignment{}, err
        }
        if current.Published() {
                return store.Assignment{}, apperr.Conflict("assignment_locked", "作业已发布，内容已锁定，不能修改").
                        WithDetails(map[string]any{"published_at": current.PublishedAt.Format(time.RFC3339)})
        }

        title, requirement, deadline, err := s.validate(in)
        if err != nil {
                return store.Assignment{}, err
        }
        if err := s.store.UpdateAssignmentDraft(ctx, id, title, requirement, deadline); err != nil {
                switch {
                case errors.Is(err, store.ErrNotFound):
                        return store.Assignment{}, apperr.NotFound("作业不存在")
                case errors.Is(err, store.ErrConflict):
                        // 并发下作业刚被发布：同样按锁定处理，并给出实际状态。
                        return store.Assignment{}, apperr.Conflict("assignment_locked", "作业已被发布，修改未生效")
                default:
                        return store.Assignment{}, err
                }
        }
        return s.store.GetAssignment(ctx, id)
}

// Publish 发布作业并生成快照。重复发布返回 409 already_published。
func (s *AssignmentService) Publish(ctx context.Context, actor store.User, id string) (store.Assignment, error) {
        if actor.Role != store.RoleTeacher {
                return store.Assignment{}, apperr.Forbidden("只有教师可以发布作业")
        }
        current, err := s.loadForActor(ctx, actor, id)
        if err != nil {
                return store.Assignment{}, err
        }
        if current.Published() {
                return store.Assignment{}, apperr.Conflict("already_published", "作业已发布，请勿重复发布").
                        WithDetails(map[string]any{"published_at": current.PublishedAt.Format(time.RFC3339)})
        }

        now := s.clk.Now()
        if err := s.store.PublishAssignment(ctx, id, actor.ID, now); err != nil {
                switch {
                case errors.Is(err, store.ErrNotFound):
                        return store.Assignment{}, apperr.NotFound("作业不存在")
                case errors.Is(err, store.ErrConflict):
                        return store.Assignment{}, apperr.Conflict("already_published", "作业已被发布，请勿重复发布")
                default:
                        return store.Assignment{}, err
                }
        }
        return s.store.GetAssignment(ctx, id)
}

// loadForActor 读取作业并做班级隔离：跨班与不存在统一 404。
func (s *AssignmentService) loadForActor(ctx context.Context, actor store.User, id string) (store.Assignment, error) {
        return loadAssignmentFor(ctx, s.store, actor, id)
}

func (s *AssignmentService) validate(in AssignmentInput) (string, string, time.Time, error) {
        title := strings.TrimSpace(in.Title)
        requirement := strings.TrimSpace(in.Requirement)
        switch {
        case title == "":
                return "", "", time.Time{}, apperr.Invalid("标题不能为空")
        case utf8.RuneCountInString(title) > maxTitleRunes:
                return "", "", time.Time{}, apperr.Invalid("标题不能超过 100 个字符")
        case requirement == "":
                return "", "", time.Time{}, apperr.Invalid("提交要求不能为空")
        case utf8.RuneCountInString(requirement) > maxRequirementRunes:
                return "", "", time.Time{}, apperr.Invalid("提交要求不能超过 2000 个字符")
        case in.DeadlineAt.IsZero():
                return "", "", time.Time{}, apperr.Invalid("截止时间不能为空")
        case !in.DeadlineAt.After(s.clk.Now()):
                return "", "", time.Time{}, apperr.Invalid("截止时间必须晚于当前业务时间")
        }
        return title, requirement, in.DeadlineAt.UTC(), nil
}

// newAssignmentID 生成作业编号（HW- + 8 位十六进制），不依赖数据库自增，
// 也不使用系统时间（业务时间必须来自注入的 clock）。
func newAssignmentID() (string, error) {
        buf := make([]byte, 4)
        if _, err := rand.Read(buf); err != nil {
                return "", fmt.Errorf("生成作业编号失败: %w", err)
        }
        return "HW-" + hex.EncodeToString(buf), nil
}

// 补交原因长度按字符数计：FR-06 要求 ≥5 且 ≤200。
const (
        minExtensionReasonRunes = 5
        maxExtensionReasonRunes = 200
)

// ExtensionService 负责"按人、一次性"的补交授权。
type ExtensionService struct {
        store *store.Store
        clk   *clock.Clock
}

// NewExtensionService 构造补交服务。
func NewExtensionService(st *store.Store, clk *clock.Clock) *ExtensionService {

}

// ExtensionInput 是补交请求体。
type ExtensionInput struct {
        StudentID  string
        DeadlineAt time.Time
        Reason     string
}

// Grant 为某学生设置一次性补交截止。
//
// 规则顺序：
//  1. 只有本班教师可批准；
//  2. 作业必须已发布（草稿没有名单，也没有可补交的提交）；
//  3. 目标必须是本班学生且在该作业名单中；
//  4. 已通过（PASSED）的学生无需补交；
//  5. 每个学生每份作业只能一次，重复 → 409 extension_exists；
//  6. 补交截止必须晚于原截止，只影响该生。
func (s *ExtensionService) Grant(ctx context.Context, actor store.User, assignmentID string, in ExtensionInput) (store.Extension, error) {
        if actor.Role != store.RoleTeacher {
                return store.Extension{}, apperr.Forbidden("只有教师可以批准补交")
        }
        a, err := loadAssignmentFor(ctx, s.store, actor, assignmentID)
        if err != nil {
                return store.Extension{}, err
        }
        if !a.Published() {
                return store.Extension{}, apperr.Conflict("not_published", "作业尚未发布，暂无学生名单可批补交")
        }

        studentID, deadline, reason, err := validateExtension(in)
        if err != nil {
                return store.Extension{}, err
        }

        target, err := s.store.GetUser(ctx, studentID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return store.Extension{}, apperr.NotFound("学生不存在")
                }
                return store.Extension{}, err
        }
        // 跨班或非学生账号一律按"不存在"处理，避免通过补交接口探测他人身份。
        if target.ClassID != actor.ClassID || target.Role != store.RoleStudent {
                return store.Extension{}, apperr.NotFound("学生不存在")
        }

        enr, err := s.store.GetEnrollment(ctx, assignmentID, studentID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return store.Extension{}, apperr.Conflict("not_enrolled", "该学生不在作业名单中").
                                WithDetails(map[string]any{"student_id": studentID})
                }
                return store.Extension{}, err
        }
        if enr.State == store.StatePassed {
                return store.Extension{}, apperr.Conflict("already_passed", "该学生已通过，无需补交").
                        WithDetails(map[string]any{"state": enr.State})
        }

        base := a.EffectiveDeadline()
        if !deadline.After(base) {
                return store.Extension{}, apperr.New(400, "extension_deadline_invalid", "补交截止必须晚于原截止时间").
                        WithDetails(map[string]any{
                                "deadline_at":          deadline.Format(time.RFC3339),
                                "original_deadline_at": base.Format(time.RFC3339),
                        })
        }

        // 先查一次给出友好冲突；并发下靠主键兜底。
        if existing, err := s.store.GetExtension(ctx, assignmentID, studentID); err == nil {
                return store.Extension{}, extensionExists(existing)
        } else if !errors.Is(err, store.ErrNotFound) {
                return store.Extension{}, err
        }

        ext := store.Extension{
                AssignmentID: assignmentID,
                StudentID:    studentID,
                DeadlineAt:   deadline,
                Reason:       reason,
                GrantedBy:    actor.ID,
                GrantedAt:    s.clk.Now(),
        }
        err = s.store.CreateExtension(ctx, store.CreateExtensionParams{
                Extension: ext,
                State:     enr.State,
                ActorRole: actor.Role,
        })
        if err != nil {
                if store.IsUniqueViolation(err) {
                        // 并发下另一个请求刚写入：按"已存在"返回，保证语义一致。
                        if existing, getErr := s.store.GetExtension(ctx, assignmentID, studentID); getErr == nil {
                                return store.Extension{}, extensionExists(existing)
                        }
                        return store.Extension{}, apperr.Conflict("extension_exists", "该学生已设置过补交").
                                WithDetails(map[string]any{"student_id": studentID})
                }
                return store.Extension{}, err
        }
        return ext, nil
}

func extensionExists(existing store.Extension) error {
        return apperr.Conflict("extension_exists", "每个学生每份作业只能设置一次补交").
                WithDetails(map[string]any{
                        "deadline_at": existing.DeadlineAt.Format(time.RFC3339),
                        "granted_at":  existing.GrantedAt.Format(time.RFC3339),
                })
}

func validateExtension(in ExtensionInput) (string, time.Time, string, error) {
        studentID := strings.TrimSpace(in.StudentID)
        reason := strings.TrimSpace(in.Reason)
        runes := utf8.RuneCountInString(reason)
        switch {
        case studentID == "":
                return "", time.Time{}, "", apperr.Invalid("student_id 不能为空")
        case in.DeadlineAt.IsZero():
                return "", time.Time{}, "", apperr.Invalid("deadline_at 不能为空：补交必须给出新的截止时间")
        case runes < minExtensionReasonRunes:
                return "", time.Time{}, "", apperr.Invalid("补交原因至少 5 个字")
        case runes > maxExtensionReasonRunes:
                return "", time.Time{}, "", apperr.Invalid("补交原因不能超过 200 个字")
        }
        return studentID, in.DeadlineAt.UTC(), reason, nil
}

// Board 返回四类清单。教师与助教都可看，但都不含正文。
func (s *OverviewService) Board(ctx context.Context, actor store.User, assignmentID string) (ClassBoard, error) {
        a, err := s.loadPublished(ctx, actor, assignmentID, true)
        if err != nil {
                return ClassBoard{}, err
        }

        rows, err := s.store.ListEnrollmentRows(ctx, a.ID)
        if err != nil {
                return ClassBoard{}, err
        }
        exts, err := s.store.ListExtensions(ctx, a.ID)
        if err != nil {
                return ClassBoard{}, err
        }
        reminders, err := s.store.ListReminders(ctx, a.ID)
        if err != nil {
                return ClassBoard{}, err
        }

        now := s.clk.Now()
        extByStudent := make(map[string]store.Extension, len(exts))
        for _, ext := range exts {
                extByStudent[ext.StudentID] = ext
        }
        reminderCount := make(map[string]int, len(rows))
        lastReminder := make(map[string]time.Time, len(rows))
        for _, r := range reminders {
                reminderCount[r.StudentID]++
                if r.CreatedAt.After(lastReminder[r.StudentID]) {
                        lastReminder[r.StudentID] = r.CreatedAt
                }
        }

        students := make([]StudentStatus, 0, len(rows))
        for _, row := range rows {
                st := StudentStatus{
                        StudentID:        row.StudentID,
                        Name:             row.StudentName,
                        State:            row.State,
                        CurrentVersionNo: row.CurrentVersionNo,
                        LastSubmittedAt:  row.LastSubmittedAt,
                        ReminderCount:    reminderCount[row.StudentID],
                        LastReminderAt:   lastReminder[row.StudentID],
                }
                st.EffectiveDeadline = a.EffectiveDeadline()
                if ext, ok := extByStudent[row.StudentID]; ok {
                        st.EffectiveDeadline = ext.DeadlineAt
                        st.ExtensionApplied = true
                }
                st.Overdue = isOverdue(st.State, st.EffectiveDeadline, now)
                students = append(students, st)
        }

        return ClassBoard{
                Assignment:     a,
                GeneratedAt:    now,
                Counts:         countsOf(students),
                CompletionRate: completionRate(students),
                Students:       students,
                Reminders:      reminders,
        }, nil
}

                return ClassSummary{}, err
        }
        exts, err := s.store.ListExtensions(ctx, a.ID)
        if err != nil {
                return ClassSummary{}, err
        }
        now := s.clk.Now()
        extByStudent := make(map[string]store.Extension, len(exts))
        for _, ext := range exts {
                extByStudent[ext.StudentID] = ext
        }

        students := make([]StudentSummary, 0, len(rows))
        for _, row := range rows {
                item := StudentSummary{
                        StudentStatus: StudentStatus{
                                StudentID:        row.StudentID,
                                Name:             row.StudentName,
                                State:            row.State,
                                CurrentVersionNo: row.CurrentVersionNo,
                                LastSubmittedAt:  row.LastSubmittedAt,
                        },
                        VersionCount:     row.SubmissionCount,
                        FirstSubmittedAt: row.FirstSubmittedAt,
                }
                item.EffectiveDeadline = a.EffectiveDeadline()
                if ext, ok := extByStudent[row.StudentID]; ok {
                        extCopy := ext
                        item.Extension = &extCopy
                        item.EffectiveDeadline = ext.DeadlineAt
                        item.ExtensionApplied = true
                }
                item.Overdue = isOverdue(item.State, item.EffectiveDeadline, now)

                // 最近一次批阅结论：只取一条用于"最终结果"，完整历史由时间线承载。
                reviews, err := s.store.ListReviews(ctx, a.ID, row.StudentID)
                if err != nil {
                        return ClassSummary{}, err
                }
                if len(reviews) > 0 {
                        last := reviews[len(reviews)-1]
                        item.LatestReview = &LatestReview{
                                Decision:  last.Decision,
                                Feedback:  last.Feedback,
                                CreatedAt: last.CreatedAt,
                        }
                }
                students = append(students, item)
        }

        entries, err := s.store.ListPointsByAssignment(ctx, a.ID)
        if err != nil {
                return ClassSummary{}, err
        }
        pointsByStudent := make(map[string]int, len(students))
        total := 0
        for _, e := range entries {
                pointsByStudent[e.StudentID] += e.Points
                total += e.Points
        }
        for i := range students {
                students[i].Points = pointsByStudent[students[i].StudentID]
        }

        events, err := s.store.ListEvents(ctx, a.ID, "")
        if err != nil {
                return ClassSummary{}, err
        }

        summary := ClassSummary{
                Assignment:     a,
                GeneratedAt:    now,
                Counts:         countsOfStatuses(students),
                CompletionRate: completionRateStatuses(students),
        }
        if actor.Role == store.RoleAssistant {
                // 助教：只读统计，不返回人均明细、奖励流水与时间线（AS-07 最小暴露）。
                summary.ReadOnlyStats = true
                return summary, nil
        }
        summary.Students = students
        summary.Points = entries
        summary.TotalPoints = total
        summary.Timeline = events
        return summary, nil
}

func isOverdue(state string, deadline, now time.Time) bool {
        // 只有"从未提交且已过有效截止"才算逾期未交；退回/待批阅有自己的类别。
        return state == store.StateNotSubmitted && now.After(deadline)
}

func countsOf(students []StudentStatus) map[string]int {
        counts := map[string]int{
                "not_submitted": 0, "submitted": 0, "returned": 0, "passed": 0, "total": len(students),
        }
        for _, st := range students {
                switch st.State {
                case store.StateNotSubmitted:
                        counts["not_submitted"]++
                case store.StateSubmitted:
                        counts["submitted"]++
                case store.StateReturned:
                        counts["returned"]++
                case store.StatePassed:
                        counts["passed"]++
                }
        }
        return counts
}

func completionRate(students []StudentStatus) float64 {
        if len(students) == 0 {
                return 0
        }
        passed := 0
        for _, st := range students {
                if st.State == store.StatePassed {
                        passed++
                }
        }
        return round3(float64(passed) / float64(len(students)))
}

func countsOfStatuses(students []StudentSummary) map[string]int {
        asStatus := make([]StudentStatus, 0, len(students))
        for _, st := range students {
                asStatus = append(asStatus, st.StudentStatus)
        }
        return countsOf(asStatus)
}

func completionRateStatuses(students []StudentSummary) float64 {
        asStatus := make([]StudentStatus, 0, len(students))
        for _, st := range students {
                asStatus = append(asStatus, st.StudentStatus)
        }
        return completionRate(asStatus)
}

func round3(v float64) float64 { return math.Round(v*1000) / 1000 }

func validateReminder(in ReminderInput) (string, string, string, error) {
        studentID := strings.TrimSpace(in.StudentID)
        channel := strings.TrimSpace(in.Channel)
        note := strings.TrimSpace(in.Note)
        if _, ok := reminderChannels[channel]; !ok {
                return "", "", "", apperr.Invalid("channel 只能是 in_app、phone 或 group")

        case studentID == "":
                return "", "", "", apperr.Invalid("student_id 不能为空")
        case note == "":

        case utf8.RuneCountInString(note) > maxReminderNoteRunes:

        }

}

// ReviewService 负责教师批阅：结论绑定到具体版本，不绑"学生当前状态"。
type ReviewService struct {
        store *store.Store
        clk   *clock.Clock
}

// NewReviewService 构造批阅服务。
func NewReviewService(st *store.Store, clk *clock.Clock) *ReviewService {
        return &ReviewService{store: st, clk: clk}
}

// ReviewInput 是批阅请求体。VersionID 必填：教师批的是"哪一版"必须明确。
type ReviewInput struct {
        VersionID string
        Decision  string
        Feedback  string
        RequestID string
}

// ReviewResult 是批阅结果。Replayed 为 true 表示同一版本同一结论的重复提交。
type ReviewResult struct {
        Review     store.Review
        Enrollment store.Enrollment
        Replayed   bool
        // PointsAwarded 表示本次批阅是否真的新发了积分（重放/非首次通过为 false）。
        PointsAwarded bool
        // Points 是本次发放的分值；未发奖为 0。
        Points int
}

// Review 批阅某个提交版本。
//
// 规则顺序：
//  1. 只有本班教师可批阅；
//  2. 作业必须已发布（草稿没有可批阅的提交）；
//  3. 同版本同结论 → 幂等返回首次结论；同版本异结论 → 409（一版一结论）；
//  4. 只能批阅"当前版本"，旧版本 → 409 stale_version；
//  5. 状态必须是 SUBMITTED，通过 → PASSED，退回 → RETURNED，并在同事务写事件。
func (s *ReviewService) Review(ctx context.Context, actor store.User, assignmentID string, in ReviewInput) (ReviewResult, error) {
        if actor.Role != store.RoleTeacher {
                return ReviewResult{}, apperr.Forbidden("只有教师可以批阅作业")
        }
        a, err := loadAssignmentFor(ctx, s.store, actor, assignmentID)
        if err != nil {
                return ReviewResult{}, err
        }
        if !a.Published() {
                return ReviewResult{}, apperr.Conflict("not_published", "作业尚未发布，没有可批阅的提交")
        }

        versionID, decision, feedback, err := validateReview(in)
        if err != nil {
                return ReviewResult{}, err
        }

        sub, err := s.store.GetSubmissionByVersionID(ctx, assignmentID, versionID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return ReviewResult{}, apperr.NotFound("提交版本不存在")
                }
                return ReviewResult{}, err
        }
        enr, err := s.store.GetEnrollment(ctx, assignmentID, sub.StudentID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return ReviewResult{}, apperr.NotFound("提交版本不存在")
                }
                return ReviewResult{}, err
        }

        // 幂等优先：已经批过的版本，重试同一结论直接返回首次结论，
        // 不因"它已不是当前版本"而误判为 stale_version。
        if existing, err := s.store.GetReviewByVersionID(ctx, versionID); err == nil {
                return replayReview(existing, decision, enr)
        } else if !errors.Is(err, store.ErrNotFound) {
                return ReviewResult{}, err
        }

        if sub.VersionNo != enr.CurrentVersionNo {
                return ReviewResult{}, apperr.Conflict("stale_version", "只能批阅当前版本，该版本已被新提交取代").
                        WithDetails(map[string]any{
                                "version_no":         sub.VersionNo,
                                "current_version_no": enr.CurrentVersionNo,
                        })
        }
        if enr.State != store.StateSubmitted {
                return ReviewResult{}, apperr.Conflict("invalid_state", "当前状态不可批阅").
                        WithDetails(map[string]any{"state": enr.State})
        }

        toState := store.StatePassed
        if decision == store.ReviewReject {
                toState = store.StateReturned
        }
        revID, err := newReviewID()
        if err != nil {
                return ReviewResult{}, err
        }
        now := s.clk.Now()
        rev := store.Review{
                ID:        revID,
                VersionID: versionID,
                Decision:  decision,
                Feedback:  feedback,
                TeacherID: actor.ID,
                RequestID: strings.TrimSpace(in.RequestID),
                CreatedAt: now,
        }

        // 通过才发奖，且与批阅同事务落库；退回、重放、非首次通过都不产生流水。
        var award *store.PointsEntry
        if toState == store.StatePassed {
                awardID, err := newPointsID()
                if err != nil {
                        return ReviewResult{}, err
                }
                award = &store.PointsEntry{
                        ID:           awardID,
                        StudentID:    sub.StudentID,
                        AssignmentID: assignmentID,
                        Reason:       store.ReasonFirstPass,
                        Points:       store.PointsFirstPass,
                        CreatedAt:    now,
                }
        }

        out, err := s.store.CreateReview(ctx, store.CreateReviewParams{
                Review:            rev,
                AssignmentID:      assignmentID,
                StudentID:         sub.StudentID,
                FromState:         enr.State,
                ToState:           toState,
                ExpectedVersionNo: enr.CurrentVersionNo,
                ActorRole:         actor.Role,
                Award:             award,
        })
        if err != nil {
                switch {
                case errors.Is(err, store.ErrConflict), store.IsUniqueViolation(err):
                        // 并发兜底：同一版本可能已被另一个请求批过。
                        if existing, getErr := s.store.GetReviewByVersionID(ctx, versionID); getErr == nil {
                                return replayReview(existing, decision, enr)
                        }
                        return ReviewResult{}, apperr.Conflict("version_conflict", "批阅状态已被并发修改，请刷新后重试")
                default:
                        return ReviewResult{}, err
                }
        }

        return ReviewResult{
                Review: rev,
                Enrollment: store.Enrollment{
                        AssignmentID:     assignmentID,
                        StudentID:        sub.StudentID,
                        State:            toState,
                        StateVersion:     enr.StateVersion + 1,
                        CurrentVersionNo: enr.CurrentVersionNo,
                },
                PointsAwarded: out.Awarded,
                Points:        pointsIf(out.Awarded),
        }, nil
}

// pointsIf 把"是否发奖"翻译成分值，避免调用方自己写 10/0 的魔法数。
func pointsIf(awarded bool) int {
        if awarded {
                return store.PointsFirstPass
        }
        return 0
}

// replayReview 处理"该版本已有结论"：同结论视为重放，异结论明确拒绝。
func replayReview(existing store.Review, decision string, enr store.Enrollment) (ReviewResult, error) {
        if existing.Decision != decision {
                return ReviewResult{}, apperr.Conflict("review_conflict", "该版本已有不同结论，不能重复批阅").
                        WithDetails(map[string]any{
                                "existing_decision":  existing.Decision,
                                "requested_decision": decision,
                        })
        }
        return ReviewResult{Review: existing, Enrollment: enr, Replayed: true}, nil
}

func validateReview(in ReviewInput) (string, string, string, error) {
        versionID := strings.TrimSpace(in.VersionID)
        decision := strings.TrimSpace(in.Decision)
        feedback := strings.TrimSpace(in.Feedback)
        switch {
        case versionID == "":
                return "", "", "", apperr.Invalid("version_id 不能为空：批阅必须指明版本")
        case decision != store.ReviewPass && decision != store.ReviewReject:
                return "", "", "", apperr.Invalid("decision 只能是 pass 或 reject")
        case decision == store.ReviewReject && feedback == "":
                return "", "", "", apperr.Invalid("退回必须填写反馈原因")
        case utf8.RuneCountInString(feedback) > maxFeedbackRunes:
                return "", "", "", apperr.Invalid("反馈不能超过 1000 个字符")
        }
        return versionID, decision, feedback, nil
}

func newReviewID() (string, error) {
        buf := make([]byte, 6)
        if _, err := rand.Read(buf); err != nil {
                return "", fmt.Errorf("生成批阅编号失败: %w", err)
        }
        return "RV-" + hex.EncodeToString(buf), nil
}

func newPointsID() (string, error) {
        buf := make([]byte, 6)
        if _, err := rand.Read(buf); err != nil {
                return "", fmt.Errorf("生成积分流水编号失败: %w", err)
        }
        return "PT-" + hex.EncodeToString(buf), nil
}

// SubmitInput 是提交请求体。
type SubmitInput struct {
        Content   string
        RequestID string
        // BaseVersion 是客户端"提交时看到的当前版本号"，首次提交传 0。
        // 为 nil 表示请求未携带，按 400 处理：不能靠默认值蒙混成"基于最新版"。
        BaseVersion *int
}

// SubmitResult 是提交成功后的返回信息。
type SubmitResult struct {
        Version           store.Submission
        Enrollment        store.Enrollment
        EffectiveDeadline time.Time
        ExtensionApplied  bool
        // Replayed 为 true 表示命中幂等重放：没有新增版本，返回的是首次结果。
        Replayed bool
}

// SubmissionHistory 是某学生在一份作业上的版本历史。
type SubmissionHistory struct {
        AssignmentID      string
        StudentID         string
        State             string
        CurrentVersionNo  int
        EffectiveDeadline time.Time
        ExtensionApplied  bool
        Versions          []store.Submission
        // Reviews 以版本号为键，包含已被新版本取代的旧结论（旧版本与意见可查）。
        Reviews map[int]store.Review
}

// Submit 追加一个提交版本。
//
// 规则顺序（先判身份与资源，再判状态与时间，最后才是写入）：
//  1. 只有本班学生本人可提交；
//  2. 作业必须已发布（草稿对学生按不存在处理）；
//  3. 已通过（PASSED）不得再提交；
//  4. 超过有效截止时间（含个人补交授权）不得提交；恰好等于截止时刻仍允许，
//     即判定条件为 now > deadline；
//  5. 幂等：同 request_id + 同内容 → 返回首次版本（Replayed=true），不新增；
//     同 request_id + 异内容 → 409 idempotency_conflict；
//  6. 旧请求保护：base_version 与服务端当前版本不一致 → 409 stale_version，
//     不写入任何内容（旧页面无法覆盖新提交）。
func (s *SubmissionService) Submit(ctx context.Context, actor store.User, assignmentID string, in SubmitInput) (SubmitResult, error) {
        if actor.Role != store.RoleStudent {
                return SubmitResult{}, apperr.Forbidden("只有学生可以提交作业")
        }
        a, err := loadAssignmentFor(ctx, s.store, actor, assignmentID)
        if err != nil {
                return SubmitResult{}, err
        }
        if !a.Published() {
                return SubmitResult{}, apperr.NotFound("作业不存在")
        }

        enr, err := s.store.GetEnrollment(ctx, assignmentID, actor.ID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return SubmitResult{}, apperr.Conflict("not_enrolled", "你不在该作业的学生名单中，无法提交")
                }
                return SubmitResult{}, err
        }
        if enr.State == store.StatePassed {
                return SubmitResult{}, apperr.Conflict("already_passed", "作业已通过，不能再次提交").
                        WithDetails(map[string]any{"state": enr.State})
        }

        content, requestID, baseVersion, err := validateSubmission(in)
        if err != nil {
                return SubmitResult{}, err
        }
        contentHash := hashContent(content)

        deadline, extended, err := s.effectiveDeadline(ctx, a, actor.ID)
        if err != nil {
                return SubmitResult{}, err
        }
        now := s.clk.Now()
        if now.After(deadline) {
                return SubmitResult{}, apperr.Conflict("deadline_passed", "已超过截止时间，不能提交").
                        WithDetails(map[string]any{
                                "effective_deadline": deadline.Format(time.RFC3339),
                                "extension_applied":  extended,
                        })
        }

        // 幂等重放：同一 request_id 已经成功过，就不再看 base_version（首次结果已生效），
        // 同内容返回首次版本，异内容明确拒绝，避免"同号改内容"悄悄覆盖。
        if existing, err := s.store.GetSubmissionByRequestID(ctx, assignmentID, actor.ID, requestID); err == nil {
                return replayResult(existing, contentHash, enr, deadline, extended)
        } else if !errors.Is(err, store.ErrNotFound) {
                return SubmitResult{}, err
        }

        // 旧请求保护：客户端看到的版本必须等于服务端当前版本，否则说明页面过期。
        if baseVersion != enr.CurrentVersionNo {
                return SubmitResult{}, apperr.Conflict("stale_version", "提交基于旧版本，请刷新后重试").
                        WithDetails(map[string]any{
                                "base_version":       baseVersion,
                                "current_version_no": enr.CurrentVersionNo,
                        })
        }

        subID, err := newSubmissionID()
        if err != nil {
                return SubmitResult{}, err
        }
        sub := store.Submission{
                ID:           subID,
                AssignmentID: assignmentID,
                StudentID:    actor.ID,
                VersionNo:    enr.CurrentVersionNo + 1,
                Content:      content,
                ContentHash:  contentHash,
                RequestID:    requestID,
                CreatedAt:    now,
        }

        err = s.store.SubmitVersion(ctx, store.SubmitVersionParams{
                Submission:        sub,
                FromState:         enr.State,
                ToState:           store.StateSubmitted,
                ExpectedVersionNo: enr.CurrentVersionNo,
                ActorRole:         actor.Role,
        })
        if err != nil {
                switch {
                case errors.Is(err, store.ErrConflict), store.IsUniqueViolation(err):
                        // 并发兜底：可能是"同一 request_id 已被另一个连接写入"，
                        // 再查一次以兑现幂等语义；否则说明版本已被人推进，让客户端刷新。
                        if existing, getErr := s.store.GetSubmissionByRequestID(ctx, assignmentID, actor.ID, requestID); getErr == nil {
                                return replayResult(existing, contentHash, enr, deadline, extended)
                        }
                        return SubmitResult{}, apperr.Conflict("version_conflict", "提交状态已被并发修改，请刷新后重试")
                default:
                        return SubmitResult{}, err
                }
        }

        return SubmitResult{
                Version: sub,
                Enrollment: store.Enrollment{
                        AssignmentID:     assignmentID,
                        StudentID:        actor.ID,
                        State:            store.StateSubmitted,
                        StateVersion:     enr.StateVersion + 1,
                        CurrentVersionNo: sub.VersionNo,
                },
                EffectiveDeadline: deadline,
                ExtensionApplied:  extended,
        }, nil
}

// History 返回某学生在一份作业上的全部版本。
//
// 学生只能查自己；教师与助教可查本班学生，跨班一律 404。
func (s *SubmissionService) History(ctx context.Context, actor store.User, assignmentID, studentID string) (SubmissionHistory, error) {
        if studentID == "" {
                studentID = actor.ID
        }
        a, err := loadAssignmentFor(ctx, s.store, actor, assignmentID)
        if err != nil {
                return SubmissionHistory{}, err
        }
        if !a.Published() && actor.Role != store.RoleTeacher {
                return SubmissionHistory{}, apperr.NotFound("作业不存在")
        }
        if err := s.ensureCanViewStudent(ctx, actor, studentID); err != nil {
                return SubmissionHistory{}, err
        }

        deadline, extended, err := s.effectiveDeadline(ctx, a, studentID)
        if err != nil {
                return SubmissionHistory{}, err
        }

        enr, err := s.store.GetEnrollment(ctx, assignmentID, studentID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return SubmissionHistory{}, apperr.NotFound("该学生不在作业名单中")
                }
                return SubmissionHistory{}, err
        }
        versions, err := s.store.ListSubmissions(ctx, assignmentID, studentID)
        if err != nil {
                return SubmissionHistory{}, err
        }
        reviews, err := s.store.ListReviews(ctx, assignmentID, studentID)
        if err != nil {
                return SubmissionHistory{}, err
        }
        byVersionNo := make(map[int]store.Review, len(reviews))
        for _, rev := range reviews {
                for _, v := range versions {
                        if v.ID == rev.VersionID {
                                byVersionNo[v.VersionNo] = rev
                                break
                        }
                }
        }
        return SubmissionHistory{
                AssignmentID:      assignmentID,
                StudentID:         studentID,
                State:             enr.State,
                CurrentVersionNo:  enr.CurrentVersionNo,
                EffectiveDeadline: deadline,
                ExtensionApplied:  extended,
                Versions:          versions,
                Reviews:           byVersionNo,
        }, nil
}

// ensureCanViewStudent 校验查看他人提交的权限：学生仅本人，教师/助教限本班学生。
func (s *SubmissionService) ensureCanViewStudent(ctx context.Context, actor store.User, studentID string) error {
        // AS-07：助教只做跟进，不能看学生答案正文；状态与时间走看板接口。
        if actor.Role == store.RoleAssistant {
                return apperr.Forbidden("助教只能查看统计信息，不可查看学生作业正文")
        }
        if actor.Role == store.RoleStudent {
                if studentID != actor.ID {
                        return apperr.Forbidden("学生只能查看自己的提交")
                }
                return nil
        }
        target, err := s.store.GetUser(ctx, studentID)
        if err != nil {
                if errors.Is(err, store.ErrNotFound) {
                        return apperr.NotFound("学生不存在")
                }
                return err
        }
        if target.ClassID != actor.ClassID || target.Role != store.RoleStudent {
                return apperr.NotFound("学生不存在")
        }
        return nil
}

// effectiveDeadline 返回该生的有效截止时间：有补交授权则取授权时间，否则取作业基准快照。
func (s *SubmissionService) effectiveDeadline(ctx context.Context, a store.Assignment, studentID string) (time.Time, bool, error) {
        ext, err := s.store.GetExtension(ctx, a.ID, studentID)
        if err == nil {
                return ext.DeadlineAt, true, nil
        }
        if !errors.Is(err, store.ErrNotFound) {
                return time.Time{}, false, err
        }
        return a.EffectiveDeadline(), false, nil
}

// replayResult 处理"request_id 已存在"：同内容视为重放，异内容视为冲突。
func replayResult(existing store.Submission, contentHash string, enr store.Enrollment, deadline time.Time, extended bool) (SubmitResult, error) {
        if existing.ContentHash != contentHash {
                return SubmitResult{}, apperr.Conflict("idempotency_conflict", "同一个 request_id 提交了不同内容").
                        WithDetails(map[string]any{"existing_version_no": existing.VersionNo})
        }
        return SubmitResult{
                Version:           existing,
                Enrollment:        enr,
                EffectiveDeadline: deadline,
                ExtensionApplied:  extended,
                Replayed:          true,
        }, nil
}

func validateSubmission(in SubmitInput) (string, string, int, error) {
        content := strings.TrimSpace(in.Content)
        requestID := strings.TrimSpace(in.RequestID)
        switch {
        case content == "":
                return "", "", 0, apperr.Invalid("提交内容不能为空")
        case utf8.RuneCountInString(content) > maxContentRunes:
                return "", "", 0, apperr.Invalid("提交内容不能超过 5000 个字符")
        case requestID == "":
                return "", "", 0, apperr.Invalid("request_id 不能为空：用于识别重复提交")
        case len(requestID) > maxRequestIDLen:
                return "", "", 0, apperr.Invalid("request_id 不能超过 64 个字符")
        case in.BaseVersion == nil:
                return "", "", 0, apperr.Invalid("base_version 不能为空：首次提交传 0，更新时传当前版本号")
        case *in.BaseVersion < 0:
                return "", "", 0, apperr.Invalid("base_version 不能为负数")
        }
        return content, requestID, *in.BaseVersion, nil
}

func hashContent(content string) string {
        sum := sha256.Sum256([]byte(content))
        return hex.EncodeToString(sum[:])
}

func newSubmissionID() (string, error) {
        buf := make([]byte, 6)
        if _, err := rand.Read(buf); err != nil {
                return "", fmt.Errorf("生成提交编号失败: %w", err)
        }
        return "SUB-" + hex.EncodeToString(buf), nil
}

// Package session 提供服务端会话。
//
// 会话只存在服务端内存中，Cookie 只携带不可猜测的随机标识（不携带用户编号），
// 因此"改请求参数冒充他人"无效：身份永远由会话解析得到。
// 进程重启会话失效，需重新登录；业务数据仍在数据库，不影响 AC-19。
package session

import (
        "crypto/rand"
        "encoding/base64"
        "sync"
        "time"

        "onlineclass/internal/clock"
)

// Session 是一条会话记录。
type Session struct {
        UserID    string
        ExpiresAt time.Time
}

// Manager 是并发安全的会话表。
type Manager struct {
        mu    sync.Mutex
        items map[string]Session
        ttl   time.Duration
        clk   *clock.Clock
}

// NewManager 创建会话管理器；ttl 为会话有效期。
func NewManager(clk *clock.Clock, ttl time.Duration) *Manager {
        return &Manager{items: make(map[string]Session), ttl: ttl, clk: clk}
}

// Create 为用户新建会话并返回会话标识。
func (m *Manager) Create(userID string) (string, error) {
        raw := make([]byte, 32)
        if _, err := rand.Read(raw); err != nil {
                return "", err
        }
        id := base64.RawURLEncoding.EncodeToString(raw)

        m.mu.Lock()
        defer m.mu.Unlock()
        m.items[id] = Session{UserID: userID, ExpiresAt: m.clk.Now().Add(m.ttl)}
        return id, nil
}

// Get 返回有效会话；过期会话被删除并视为不存在。
func (m *Manager) Get(id string) (Session, bool) {
        if id == "" {
                return Session{}, false
        }
        m.mu.Lock()
        defer m.mu.Unlock()
        s, ok := m.items[id]
        if !ok {
                return Session{}, false
        }
        if !s.ExpiresAt.After(m.clk.Now()) {
                delete(m.items, id)
                return Session{}, false
        }
        return s, true
}

// Destroy 删除会话（退出登录）。
func (m *Manager) Destroy(id string) {
        m.mu.Lock()
        defer m.mu.Unlock()
        delete(m.items, id)
}

// Sweep 清理已过期会话，返回清理数量。
func (m *Manager) Sweep() int {
        m.mu.Lock()
        defer m.mu.Unlock()
        now := m.clk.Now()
        n := 0
        for id, s := range m.items {
                if !s.ExpiresAt.After(now) {
                        delete(m.items, id)
                        n++
                }
        }
        return n
}

// Len 返回当前会话数（测试与运维观测用）。
func (m *Manager) Len() int {
        m.mu.Lock()
        defer m.mu.Unlock()
        return len(m.items)
}

// UpdateAssignmentDraft 只更新草稿；已发布返回 ErrConflict（调用方转 409 assignment_locked）。
//
// 条件写在 WHERE 里，避免"先查后写"的竞态：并发发布与修改只会有一个成功。
func (s *Store) UpdateAssignmentDraft(ctx context.Context, id, title, requirement string, deadline time.Time) error {
        res, err := s.db.ExecContext(ctx,
                `UPDATE assignments SET title = ?, requirement = ?, deadline_at = ?
                 WHERE id = ? AND state = 'DRAFT'`,
                title, requirement, formatTime(deadline), id)
        if err != nil {
                return fmt.Errorf("更新草稿失败: %w", err)
        }
        n, err := res.RowsAffected()
        if err != nil {
                return fmt.Errorf("读取更新结果失败: %w", err)
        }
        if n == 0 {
                var state string
                err := s.db.QueryRowContext(ctx, `SELECT state FROM assignments WHERE id = ?`, id).Scan(&state)
                if err == sql.ErrNoRows {
                        return ErrNotFound
                }
                if err != nil {
                        return fmt.Errorf("查询作业状态失败: %w", err)
                }
                return ErrConflict
        }
        return nil
}

// PublishAssignment 在同一事务内完成：状态置为 PUBLISHED、把当前内容写入快照、为该班学生建状态行。
//
// 快照取自表的 title/requirement/deadline_at 列（而不是调用方传参），
// 保证快照与发布瞬间的草稿内容逐字一致；并发重复发布只有一次成功，其余返回 ErrConflict。
func (s *Store) PublishAssignment(ctx context.Context, id, by string, at time.Time) error {
        return s.Tx(ctx, func(tx *sql.Tx) error {
                res, err := tx.ExecContext(ctx,
                        `UPDATE assignments
                         SET state = 'PUBLISHED',
                             published_at = ?,
                             snapshot_title = title,
                             snapshot_requirement = requirement,
                             snapshot_deadline_at = deadline_at,
                             snapshot_by = ?,
                             snapshot_at = ?
                         WHERE id = ? AND state = 'DRAFT'`,
                        formatTime(at), by, formatTime(at), id)
                if err != nil {
                        return fmt.Errorf("发布作业失败: %w", err)
                }
                n, err := res.RowsAffected()
                if err != nil {
                        return fmt.Errorf("读取发布结果失败: %w", err)
                }
                if n == 0 {
                        var state string
                        err := tx.QueryRowContext(ctx, `SELECT state FROM assignments WHERE id = ?`, id).Scan(&state)
                        if err == sql.ErrNoRows {
                                return ErrNotFound
                        }
                        if err != nil {
                                return fmt.Errorf("查询作业状态失败: %w", err)
                        }
                        return ErrConflict
                }

                // 发布即固化班级花名册：此后新入班学生不追溯该作业，看板/汇总以状态行为准。
                if _, err := tx.ExecContext(ctx,
                        `INSERT OR IGNORE INTO enrollments (assignment_id, student_id, state, state_version, current_version_no)
                         SELECT ?, id, 'NOT_SUBMITTED', 0, 0
                         FROM users
                         WHERE role = 'student' AND class_id = (SELECT class_id FROM assignments WHERE id = ?)`,
                        id, id); err != nil {
                        return fmt.Errorf("创建学生作业状态失败: %w", err)
                }
                return nil
        })
}

func (s *Store) Migrate() error {
        ctx := context.Background()
        if _, err := s.db.ExecContext(ctx, `CREATE TABLE IF NOT EXISTS schema_migrations (
                version TEXT PRIMARY KEY,
                applied_at TEXT NOT NULL
        )`); err != nil {
                return fmt.Errorf("创建迁移记录表失败: %w", err)
        }

        entries, err := fs.Glob(migrationFS, "migrations/*.sql")
        if err != nil {
                return fmt.Errorf("读取迁移文件失败: %w", err)
        }
        sort.Strings(entries)

        for _, name := range entries {
                var applied int
                if err := s.db.QueryRowContext(ctx,
                        `SELECT COUNT(1) FROM schema_migrations WHERE version = ?`, name).Scan(&applied); err != nil {
                        return fmt.Errorf("查询迁移状态失败: %w", err)
                }
                if applied > 0 {
                        continue
                }
                body, err := migrationFS.ReadFile(name)
                if err != nil {
                        return fmt.Errorf("读取迁移 %s 失败: %w", name, err)
                }
                if err := s.applyMigration(ctx, name, string(body)); err != nil {
                        return err
                }
        }
        return nil
}

// applyMigration 在单个事务内执行迁移脚本并登记版本。
func (s *Store) applyMigration(ctx context.Context, name, body string) error {
        tx, err := s.db.BeginTx(ctx, nil)
        if err != nil {
                return fmt.Errorf("开启迁移事务失败: %w", err)
        }
        if _, err := tx.ExecContext(ctx, body); err != nil {
                _ = tx.Rollback()
                return fmt.Errorf("执行迁移 %s 失败: %w", name, err)
        }
        if _, err := tx.ExecContext(ctx,
                `INSERT INTO schema_migrations (version, applied_at) VALUES (?, strftime('%Y-%m-%dT%H:%M:%fZ','now'))`,
                name); err != nil {
                _ = tx.Rollback()
                return fmt.Errorf("登记迁移 %s 失败: %w", name, err)
        }
        if err := tx.Commit(); err != nil {
                return fmt.Errorf("提交迁移 %s 失败: %w", name, err)
        }
        return nil
}

const timeLayout = time.RFC3339Nano

func parseTime(s string) time.Time {
        if s == "" {
                return time.Time{}
        }
        for _, layout := range []string{timeLayout, time.RFC3339} {
                if t, err := time.Parse(layout, s); err == nil {
                        return t.UTC()
                }
        }
        return time.Time{}
}

func formatTime(t time.Time) string {
        if t.IsZero() {
                return ""
        }
        return t.UTC().Format(timeLayout)
}

        ToState           string

        Award *PointsEntry

// CreateReviewResult 回报本次批阅是否真的新发了积分。
type CreateReviewResult struct {
        Awarded bool
}

// CreateReview 在同一事务内：写批阅结论、推进状态行（乐观锁）、写审计事件。
//
// 只有"当前版本 + 状态为 SUBMITTED"才能成功；否则条件更新影响 0 行返回 ErrConflict。
// 同一版本第二个结论由 UNIQUE(version_id) 拦截。
func (s *Store) CreateReview(ctx context.Context, p CreateReviewParams) (CreateReviewResult, error) {
        var awarded bool
        err := s.Tx(ctx, func(tx *sql.Tx) error {
                rev := p.Review
                var reqID any
                if rev.RequestID != "" {
                        reqID = rev.RequestID
                }
                if _, err := tx.ExecContext(ctx,
                        `INSERT INTO reviews (id, version_id, decision, feedback, teacher_id, request_id, created_at)
                         VALUES (?, ?, ?, ?, ?, ?, ?)`,
                        rev.ID, rev.VersionID, rev.Decision, rev.Feedback, rev.TeacherID, reqID,
                        formatTime(rev.CreatedAt)); err != nil {
                        return fmt.Errorf("写入批阅结论失败: %w", err)
                }

                res, err := tx.ExecContext(ctx,
                        `UPDATE enrollments
                         SET state = ?, state_version = state_version + 1
                         WHERE assignment_id = ? AND student_id = ? AND current_version_no = ? AND state = ?`,
                        p.ToState, p.AssignmentID, p.StudentID, p.ExpectedVersionNo, p.FromState)
                if err != nil {
                        return fmt.Errorf("推进批阅状态失败: %w", err)
                }
                n, err := res.RowsAffected()
                if err != nil {
                        return fmt.Errorf("读取状态更新结果失败: %w", err)
                }
                if n == 0 {
                        return ErrConflict
                }

                label := "批阅通过"
                if rev.Decision == ReviewReject {
                        label = "批阅退回"
                }
                detail := label
                if rev.Feedback != "" {
                        detail = fmt.Sprintf("%s：%s", label, rev.Feedback)
                }
                if _, err := tx.ExecContext(ctx,
                        `INSERT INTO events
                         (assignment_id, student_id, actor_id, actor_role, type, detail, from_state, to_state, version_no, created_at)
                         VALUES (?, ?, ?, ?, 'reviewed', ?, ?, ?, ?, ?)`,
                        p.AssignmentID, p.StudentID, rev.TeacherID, p.ActorRole, detail,
                        p.FromState, p.ToState, p.ExpectedVersionNo, formatTime(rev.CreatedAt)); err != nil {
                        return fmt.Errorf("写入批阅事件失败: %w", err)
                }

                if p.Award != nil {
                        res, err := tx.ExecContext(ctx,
                                `INSERT INTO points_ledger (id, student_id, assignment_id, reason, points, created_at)
                                 VALUES (?, ?, ?, ?, ?, ?)
                                 ON CONFLICT (assignment_id, student_id, reason) DO NOTHING`,
                                p.Award.ID, p.Award.StudentID, p.Award.AssignmentID, p.Award.Reason,
                                p.Award.Points, formatTime(p.Award.CreatedAt))
                        if err != nil {
                                return fmt.Errorf("写入积分流水失败: %w", err)
                        }
                        n, err := res.RowsAffected()
                        if err != nil {
                                return fmt.Errorf("读取发奖结果失败: %w", err)
                        }
                        // 影响 1 行 = 本次真的发奖；0 行 = 之前已发过，保持幂等。
                        awarded = n == 1
                }
                return nil
        })
        return CreateReviewResult{Awarded: awarded}, err
}

        "context"
        "database/sql"
        "fmt"
        "net/url"
        "os"
        "path/filepath"

        _ "modernc.org/sqlite"
)

// Store 持有数据库连接池。
type Store struct {
        db *sql.DB
}

// Open 打开（必要时创建）数据库文件并设置连接参数。
func Open(path string) (*Store, error) {
        if dir := filepath.Dir(path); dir != "" && dir != "." {
                if err := os.MkdirAll(dir, 0o755); err != nil {
                        return nil, fmt.Errorf("创建数据库目录失败: %w", err)
                }
        }
        dsn := "file:" + url.PathEscape(filepath.ToSlash(path)) +
                "?_pragma=journal_mode(WAL)" +
                "&_pragma=busy_timeout(5000)" +
                "&_pragma=foreign_keys(1)" +
                "&_txlock=immediate"
        db, err := sql.Open("sqlite", dsn)
        if err != nil {
                return nil, fmt.Errorf("打开数据库失败: %w", err)
        }
        // 写操作串行化交给 _txlock=immediate + busy_timeout，连接数保持小值即可。
        db.SetMaxOpenConns(4)
        db.SetMaxIdleConns(4)
        if err := db.PingContext(context.Background()); err != nil {
                db.Close()
                return nil, fmt.Errorf("连接数据库失败: %w", err)
        }
        return &Store{db: db}, nil
}

// DB 返回底层连接，供各 repository 使用。
func (s *Store) DB() *sql.DB { return s.db }

// Close 关闭数据库。
func (s *Store) Close() error {
        if s.db == nil {
                return nil
        }
        return s.db.Close()
}

// Tx 在事务中执行 fn；fn 返回错误时整体回滚。
func (s *Store) Tx(ctx context.Context, fn func(tx *sql.Tx) error) error {
        tx, err := s.db.BeginTx(ctx, nil)
        if err != nil {
                return fmt.Errorf("开启事务失败: %w", err)
        }
        if err := fn(tx); err != nil {
                _ = tx.Rollback()
                return err
        }
        if err := tx.Commit(); err != nil {
                return fmt.Errorf("提交事务失败: %w", err)
        }
        return nil
}

// GetEnrollment 读取学生作业状态行；不存在返回 ErrNotFound。
func (s *Store) GetEnrollment(ctx context.Context, assignmentID, studentID string) (Enrollment, error) {
        var e Enrollment
        err := s.db.QueryRowContext(ctx,
                `SELECT assignment_id, student_id, state, state_version, current_version_no
                 FROM enrollments WHERE assignment_id = ? AND student_id = ?`,
                assignmentID, studentID).
                Scan(&e.AssignmentID, &e.StudentID, &e.State, &e.StateVersion, &e.CurrentVersionNo)
        if err != nil {
                if err == sql.ErrNoRows {
                        return Enrollment{}, ErrNotFound
                }
                return Enrollment{}, fmt.Errorf("查询学生作业状态失败: %w", err)
        }
        return e, nil
}

// GetExtension 读取补交授权；不存在返回 ErrNotFound。
func (s *Store) GetExtension(ctx context.Context, assignmentID, studentID string) (Extension, error) {
        var (
                ext             Extension
                deadline, grant string
        )
        err := s.db.QueryRowContext(ctx,
                `SELECT assignment_id, student_id, deadline_at, reason, granted_by, granted_at
                 FROM extensions WHERE assignment_id = ? AND student_id = ?`,
                assignmentID, studentID).
                Scan(&ext.AssignmentID, &ext.StudentID, &deadline, &ext.Reason, &ext.GrantedBy, &grant)
        if err != nil {
                if err == sql.ErrNoRows {
                        return Extension{}, ErrNotFound
                }
                return Extension{}, fmt.Errorf("查询补交授权失败: %w", err)
        }
        ext.DeadlineAt = parseTime(deadline)
        ext.GrantedAt = parseTime(grant)
        return ext, nil
}

// GetSubmissionByRequestID 按请求号查询已存在的版本，用于识别重复请求。
func (s *Store) GetSubmissionByRequestID(ctx context.Context, assignmentID, studentID, requestID string) (Submission, error) {
        var (
                sub     Submission
                created string
        )
        err := s.db.QueryRowContext(ctx,
                `SELECT id, assignment_id, student_id, version_no, content, content_hash, request_id, created_at
                 FROM submission_versions
                 WHERE assignment_id = ? AND student_id = ? AND request_id = ?`,

                Scan(&sub.ID, &sub.AssignmentID, &sub.StudentID, &sub.VersionNo, &sub.Content,
                        &sub.ContentHash, &sub.RequestID, &created)
        if err != nil {
                if err == sql.ErrNoRows {
                        return Submission{}, ErrNotFound
                }
                return Submission{}, fmt.Errorf("查询提交请求失败: %w", err)
        }
        sub.CreatedAt = parseTime(created)
        return sub, nil
}

// ListSubmissions 返回某学生某作业的全部版本，按版本号升序（含历史版本）。
func (s *Store) ListSubmissions(ctx context.Context, assignmentID, studentID string) ([]Submission, error) {
        rows, err := s.db.QueryContext(ctx,
                `SELECT id, assignment_id, student_id, version_no, content, content_hash, request_id, created_at
                 FROM submission_versions
                 WHERE assignment_id = ? AND student_id = ?
                 ORDER BY version_no ASC`,
                assignmentID, studentID)
        if err != nil {
                return nil, fmt.Errorf("查询提交版本失败: %w", err)
        }
        defer rows.Close()

        out := make([]Submission, 0, 2)
        for rows.Next() {
                var (
                        sub     Submission
                        created string
                )
                if err := rows.Scan(&sub.ID, &sub.AssignmentID, &sub.StudentID, &sub.VersionNo, &sub.Content,
                        &sub.ContentHash, &sub.RequestID, &created); err != nil {
                        return nil, fmt.Errorf("读取提交版本失败: %w", err)
                }
                sub.CreatedAt = parseTime(created)
                out = append(out, sub)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("遍历提交版本失败: %w", err)
        }
        return out, nil
}

// SubmitVersionParams 是一次提交落库所需的全部信息。
type SubmitVersionParams struct {
        Submission        Submission
        FromState         string
        ToState           string
        ExpectedVersionNo int
        ActorRole         string
}

// SubmitVersion 在同一事务内：追加版本、推进状态行（乐观锁）、写审计事件。
//
// 并发保护有两层：
//  1. UNIQUE(assignment, student, version_no) 与 UNIQUE(..., request_id) 兜底重复插入；
//  2. UPDATE ... WHERE current_version_no = 期望值 保证状态行只被推进一次。
//
// 任一条件不满足返回 ErrConflict，由业务层转 409，绝不静默覆盖。
func (s *Store) SubmitVersion(ctx context.Context, p SubmitVersionParams) error {
        return s.Tx(ctx, func(tx *sql.Tx) error {
                sub := p.Submission
                if _, err := tx.ExecContext(ctx,
                        `INSERT INTO submission_versions
                         (id, assignment_id, student_id, version_no, content, content_hash, request_id, created_at)
                         VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
                        sub.ID, sub.AssignmentID, sub.StudentID, sub.VersionNo, sub.Content,
                        sub.ContentHash, sub.RequestID, formatTime(sub.CreatedAt)); err != nil {
                        return fmt.Errorf("写入提交版本失败: %w", err)
                }

                res, err := tx.ExecContext(ctx,
                        `UPDATE enrollments
                         SET state = ?, current_version_no = ?, state_version = state_version + 1
                         WHERE assignment_id = ? AND student_id = ? AND current_version_no = ? AND state = ?`,
                        p.ToState, sub.VersionNo, sub.AssignmentID, sub.StudentID, p.ExpectedVersionNo, p.FromState)
                if err != nil {
                        return fmt.Errorf("推进作业状态失败: %w", err)
                }
                n, err := res.RowsAffected()
                if err != nil {
                        return fmt.Errorf("读取状态更新结果失败: %w", err)
                }
                if n == 0 {
                        return ErrConflict
                }

                if _, err := tx.ExecContext(ctx,
                        `INSERT INTO events
                         (assignment_id, student_id, actor_id, actor_role, type, detail, from_state, to_state, version_no, created_at)
                         VALUES (?, ?, ?, ?, 'submitted', ?, ?, ?, ?, ?)`,
                        sub.AssignmentID, sub.StudentID, sub.StudentID, p.ActorRole,
                        fmt.Sprintf("提交第 %d 版", sub.VersionNo), p.FromState, p.ToState, sub.VersionNo,
                        formatTime(sub.CreatedAt)); err != nil {
                        return fmt.Errorf("写入审计事件失败: %w", err)
                }
                return nil
        })
}

// IsUniqueViolation 判断错误是否由唯一约束冲突导致。
//
// 业务层用它把"并发下同版本重复插入"翻译成 409，而不是 500。
func IsUniqueViolation(err error) bool {
        if err == nil {
                return false
        }
        var se interface{ Code() int }
        if errors.As(err, &se) {
                // modernc/sqlite：2067 = SQLITE_CONSTRAINT_UNIQUE，1555 = 主键冲突
                if se.Code() == 2067 || se.Code() == 1555 {
                        return true
                }
        }
        return strings.Contains(err.Error(), "UNIQUE constraint failed")
}

// GetUser 按编号查询账号；不存在返回 ErrNotFound。
func (s *Store) GetUser(ctx context.Context, id string) (User, error) {
        var u User
        err := s.db.QueryRowContext(ctx,
                `SELECT id, class_id, name, role, password_hash FROM users WHERE id = ?`, id).
                Scan(&u.ID, &u.ClassID, &u.Name, &u.Role, &u.PasswordHash)
        if err != nil {
                if err == sql.ErrNoRows {
                        return User{}, ErrNotFound
                }
                return User{}, fmt.Errorf("查询账号失败: %w", err)
        }
        return u, nil
}

// ListStudents 返回指定班级的学生，按编号排序（汇总与看板使用）。
func (s *Store) ListStudents(ctx context.Context, classID string) ([]User, error) {
        rows, err := s.db.QueryContext(ctx,
                `SELECT id, class_id, name, role, password_hash FROM users
                 WHERE class_id = ? AND role = 'student' ORDER BY id`, classID)
        if err != nil {
                return nil, fmt.Errorf("查询学生列表失败: %w", err)
        }
        defer rows.Close()

        var out []User
        for rows.Next() {
                var u User
                if err := rows.Scan(&u.ID, &u.ClassID, &u.Name, &u.Role, &u.PasswordHash); err != nil {
                        return nil, fmt.Errorf("读取学生失败: %w", err)
                }
                out = append(out, u)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("遍历学生失败: %w", err)
        }
        return out, nil
}
